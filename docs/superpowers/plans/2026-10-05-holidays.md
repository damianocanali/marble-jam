# Holiday Packs Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Halloween, Thanksgiving and Christmas packs (songs + drawn backgrounds) that switch on by date, with a menu banner, a holiday shelf in My Songs and a DEBUG preview.

**Architecture:** `Seasons.swift` holds the catalogue (windows, banners, `SongRecipe` melodies) and the pure `active(on:calendar:)`. `SongStore.addCopy(of:)` saves a shelf song as the player's own. Backgrounds get an optional `season` in the manifest; `BackgroundLibrary.available` filters by season. UI: RootView computes the season and passes it to MenuView/LibraryView.

**Tech Stack:** Swift 5.9, SwiftUI, Core Graphics (background tool), XCTest, XcodeGen, iOS 17.

**Spec:** `docs/superpowers/specs/2026-10-05-holidays-design.md`

## Global Constraints

- Windows (inclusive): Halloween Oct 1–Nov 1; Thanksgiving Nov 2–day after the 4th Thursday of November; Christmas Dec 1–Jan 6.
- Banners: "🎃 Halloween is here!", "🦃 Happy Thanksgiving!", "🎄 Merry Christmas!".
- Season ids: `halloween`, `thanksgiving`, `christmas`. Background files `NN-<season>-<name>.png` → manifest `"season"`.
- Melody notes: bar note index 0…14 (C4…C6 white keys); beats in multiples of 0.5; every note must land on the beat.
- DEBUG-only preview stored in `UserDefaults` key `season.preview`: `""` = by date, `"off"` = none, else a season id.
- Copy naming: recipe name, then "<name> copy", "<name> copy 2", ...
- No stock art; songs public domain (see spec table).
- Test command: `xcodegen -q && xcodebuild test -project MarbleJam.xcodeproj -scheme MarbleJam -destination 'platform=iOS Simulator,name=iPhone 17' 2>&1 | grep -E "error:|Executed [0-9]+ tests, with|TEST (SUCC|FAIL)" | sort -u`

## Review Focus

1. Christmas spans New Year: Jan 1–6 must be Christmas, Jan 7 nothing — `testChristmasCrossesNewYear` (Task 1).
2. Thanksgiving moves every year — 2026/2027/2028 dates and the day after — `testThanksgivingFollowsTheFourthThursday` (Task 1).
3. A melody that can't be laid out on the beat must fail before release — `testEveryHolidaySongIsAllOnTheBeat` (Task 1).
4. The player's chosen background going off-season must not crash or show a missing image — `testOffSeasonChoiceFallsBackToNight` (Task 3).
5. Taking the same holiday song twice must not overwrite the first copy — `testTakingAShelfSongTwiceMakesACopy` (Task 2).

```swift
import Foundation

/// A ready-made song: a public-domain melody laid out as a course. Notes are bar note indexes (0...14 = C4...C6, white keys);
/// `beats` is the time since the previous note.
struct SongRecipe {
    let name: String
    let instrument: Instrument
    let melody: [(beats: Double, note: Int)]

    func course() -> Course {
        var c = Course()
        for m in melody { _ = Engine.smartAdd(&c, beats: m.beats, note: m.note) }
        return c
    }
}

/// A holiday pack. Its window comes round every year.

}

    /// The season whose window contains `date` (windows starting this year or, for ones crossing New Year, last year).
    static func active(on date: Date, calendar: Calendar) -> Season? {
        let year = calendar.component(.year, from: date), today = calendar.startOfDay(for: date)
        for s in all {
            for y in [year - 1, year] {
                let w = s.window(y, calendar)
                if today >= calendar.startOfDay(for: w.first) && today <= calendar.startOfDay(for: w.last) { return s }
            }
        }
        return nil
    }

    /// The live season, unless a DEBUG preview overrides it: "" = by date, "off" = none, otherwise a season id.
    static func current(on date: Date, preview: String, calendar: Calendar = .current) -> Season? {
        switch preview {
        case "": active(on: date, calendar: calendar)
        case "off": nil
        default: all.first { $0.id == preview }
        }
    }
}
```

```swift
    /// `base`, or "<base> copy", "<base> copy 2", ... whichever is free.
    private func uniqueName(_ base: String, startWithCopy: Bool) -> String {
        let names = Set(all().map(\.name))
        if !startWithCopy && !names.contains(base) { return base }
        var name = "\(base) copy", k = 2
        while names.contains(name) { name = "\(base) copy \(k)"; k += 1 }
        return name
    }

    func duplicate(_ id: UUID) -> Song? {
        guard let s = song(id) else { return nil }
        return try? save(Song(name: uniqueName(s.name, startWithCopy: true), course: s.course, instrument: s.instrument))
    }

    /// Saves a ready-made song (from a holiday shelf) as the player's own.
    func addCopy(of template: Song) -> Song {
        let s = Song(name: uniqueName(template.name, startWithCopy: false), course: template.course, instrument: template.instrument)
        return (try? save(s)) ?? s
    }
```

```swift
    static func available(_ all: [BackgroundOption], season: String?) -> [BackgroundOption] {
        all.filter { $0.season != nil && $0.season == season } + all.filter { $0.season == nil }
    }

    /// What to show: the player's choice if it is available now; with no choice, the season's first background; else nil (night sky).
    static func resolve(_ all: [BackgroundOption], chosenID: String, season: String?) -> BackgroundOption? {
        let now = available(all, season: season)
        if chosenID.isEmpty { return season == nil ? nil : now.first { $0.season == season } }
        return now.first { $0.id == chosenID }
    }
```

```swift
/// A pumpkin sitting with its base at (x, y); `face` carves a glowing jack-o'-lantern.
func pumpkin(_ ctx: CGContext, _ x: CGFloat, _ y: CGFloat, _ w: CGFloat, face: Bool) {
    let h = w * 0.72
    ctx.saveGState()
    ctx.setShadow(offset: CGSize(width: 0, height: 8), blur: 16, color: cg((0, 0, 0), 0.5))
    for k in [-0.32, 0.32, 0.0] as [CGFloat] {                                   // three lobes, middle on top
        ctx.setFillColor(cg(k == 0 ? rgb(0xF08A24) : rgb(0xD96F14)))
        ctx.fillEllipse(in: CGRect(x: x - w * 0.36 + k * w, y: y - h, width: w * 0.72, height: h))
    }
    ctx.restoreGState()
    ctx.setStrokeColor(cg(rgb(0x9A4A0A), 0.6)); ctx.setLineWidth(w * 0.025)
    for k in [-0.18, 0.18] as [CGFloat] {
        ctx.move(to: CGPoint(x: x + k * w, y: y - h * 0.95)); ctx.addQuadCurve(to: CGPoint(x: x + k * w, y: y - h * 0.05), control: CGPoint(x: x + k * w * 1.6, y: y - h / 2))
    }
    ctx.strokePath()
    ctx.setFillColor(cg(rgb(0x3E6B2A))); ctx.fill(CGRect(x: x - w * 0.04, y: y - h - w * 0.14, width: w * 0.08, height: w * 0.18))
    if face {
        ctx.saveGState(); ctx.setShadow(offset: .zero, blur: w * 0.15, color: cg(rgb(0xFFD23F)))
        ctx.setFillColor(cg(rgb(0xFFD23F)))
        for sx in [-1.0, 1.0] as [CGFloat] {
            let e = CGMutablePath(); e.move(to: CGPoint(x: x + sx * w * 0.22, y: y - h * 0.62))
            e.addLine(to: CGPoint(x: x + sx * w * 0.08, y: y - h * 0.62)); e.addLine(to: CGPoint(x: x + sx * w * 0.15, y: y - h * 0.78)); e.closeSubpath()

        ctx.addPath(m); ctx.fillPath(); ctx.restoreGState()
    }
}

/// Night sky (top to bottom colours), a big moon and stars: shared by the Halloween scenes.
func spookySky(_ c: Canvas, top: RGB, bottom: RGB, moonX: CGFloat, moonY: CGFloat, seed: Int32) {
    c.shade { x, y in
        let t = Float(y) / Float(H)
        return add(mix(top, bottom, t), scale((0.25, 0.2, 0.35), fbm(Float(x) / 300, Float(y) / 300, seed, 4) * 0.25 * (1 - t)))

func purpleFog(_ c: Canvas, seed: Int32) {

/// A simple maple-like leaf at (x, y), size s, turned by a, with a centre vein.
func leaf(_ ctx: CGContext, _ x: CGFloat, _ y: CGFloat, _ s: CGFloat, _ a: CGFloat, _ color: RGB) {
    ctx.saveGState(); ctx.translateBy(x: x, y: y); ctx.rotate(by: a)
    let p = CGMutablePath()
    for k in 0..<10 {
        let ang = CGFloat(k) * .pi / 5 - .pi / 2, rr = k % 2 == 0 ? s : s * 0.5
        let pt = CGPoint(x: cos(ang) * rr, y: sin(ang) * rr)
        k == 0 ? p.move(to: pt) : p.addLine(to: pt)
    }
    p.closeSubpath()
    ctx.setShadow(offset: CGSize(width: 4, height: 6), blur: 8, color: cg((0, 0, 0), 0.3))
    ctx.addPath(p); ctx.setFillColor(cg(color)); ctx.fillPath()
    ctx.setShadow(offset: .zero, blur: 0, color: nil)
    ctx.setStrokeColor(cg(scale(color, 0.6))); ctx.setLineWidth(s * 0.06)
    ctx.move(to: CGPoint(x: 0, y: s * 0.9)); ctx.addLine(to: CGPoint(x: 0, y: -s * 0.7)); ctx.strokePath()
    ctx.restoreGState()
}

func cornfieldSunset(_ c: Canvas, seed: Int32) {

            let col = cg(mix(rgb(0x3A2410), rgb(0x120A04), Float(row) / 3))
```
